import os
import sys
import torch
from ultralytics import YOLO

def run_training():
    # --- 1. HARDWARE LOCK ---
    if not torch.cuda.is_available():
        sys.exit("❌ CRITICAL ERROR: NVIDIA GPU not detected. Training aborted.")
    
    device = 0
    gpu_name = torch.cuda.get_device_name(0)
    print(f"✅ HARDWARE LOCK ENGAGED: {gpu_name}")

    # --- 2. CONFIGURATION ---
    model_name = '/home/shatadal/SAMDEF/apps/brain_python/src/models/yolo26s.pt'
    data_yaml = '/home/shatadal/SAMDEF/raw_data/processed_tiles/data.yaml'
    project_name = 'SAMDEF_ISR'
    run_name = 'Run1_Stage1'  

    # Verify weights exist before starting to avoid crashing later
    if not os.path.exists(model_name):
        sys.exit(f"❌ ERROR: Checkpoint not found at {model_name}. Please verify path.")

    model = YOLO(model_name)

    # --- 3. ENGAGE TRAINING ---
    print(f"INITIALIZING : {run_name}")
    print("   Strategy: Stage 1")

    results = model.train(
        data=data_yaml,
        
        # --- CORE TRAINING ---
        epochs=100, 
        imgsz=896,
        batch=4,
        nbs=64,
        amp=True,
        cache=False,
        
        # --- OPTIMIZER & LR (STABLE, NOT AGGRESSIVE) ---
        optimizer='MuSGD',
        lr0=0.01,            # 🔒 Stable start (not too fast, not too slow)
        lrf=0.2,
        momentum=0.937,
        weight_decay=0.0005,
        cos_lr=True,
        warmup_epochs=5,
        #warmup_momentum=0.6,
        
        # --- LOSS (GEOMETRY-FOCUSED, STABLE) ---
        box=7.5,
        cls=0.5,
        
         # --- AUGMENTATION (PROGRESSIVE MOSAIC) ---
        mosaic=0.4,           # Epoch 0–50: full diversity
        scale=0.0,
        translate=0.05,
        fliplr=0.5,
        flipud=0.5,
        degrees=0.0,
        shear=0.0,
        perspective=0.0,
        close_mosaic=0,

          # --- SYSTEM ---
        workers=8,            # Lower than 12 to avoid CPU jitter
        plots=False,
        save=True,
        save_period=10,
        seed=0,
        deterministic=True,

        # --- LOGGING & EXECUTION ---
        project=project_name,
        name=run_name,
        exist_ok=True,
        verbose=True,
        device=device,
        resume=False          # 🔒 Must be False to apply new schedule
    )

    print("-" * 50)
    print("STAGE 1 COMPLETE")
    print(f"Weights: {project_name}/{run_name}/weights/best.pt")

if __name__ == '__main__':
    run_training()
